cmake_minimum_required(VERSION 3.12)
project(cachesim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(cachelib src/cache.cpp)
target_include_directories(cachelib PUBLIC include)
target_compile_options(cachelib PRIVATE -Wall -Wextra -Wpedantic)
add_executable(cachesim src/main.cpp)
target_link_libraries(cachesim PRIVATE cachelib)
enable_testing()
add_executable(cache_tests tests/test_cache.cpp)
target_link_libraries(cache_tests PRIVATE cachelib)
add_test(NAME cache_tests COMMAND cache_tests)
